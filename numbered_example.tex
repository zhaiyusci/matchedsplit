\documentclass{article}
\usepackage[overridesplit,overridealign]{matchedsplit}
\begin{document}
\section*{Automatic brackets, numbered rows}
\begin{align}
F &= \left( a+\left[\frac{x}{y}+b \label{eq:first}\\
  &\quad +c\right]+d \label{eq:second}\\
  &\quad +\frac{p}{q}\right)\label{eq:third}
\end{align}
Each row has its own number: \eqref{eq:first}, \eqref{eq:second},
\eqref{eq:third}. Both delimiter pairs match across rows.

\section*{One number for the entire split}
\begin{equation}
\begin{split}
G &= \left(\frac{a}{b}\\
  &\quad +c\right)
\end{split}
\end{equation}
